#!/bin/bash # Author: Jaco De Villiers # Trash Cleaner - user level page. # - Default view: scans and shows what is in the trash (nothing is deleted). # - action=clean&token=...: empties every trash location and reports what was freed. PLUGIN_DIR="$(cd "$(dirname "$0")" && pwd)" . "$PLUGIN_DIR/lib.sh" ACTION="$(get_param action)" TOKEN="$(get_param token)" page_style echo '
' echo "

$BROOM_SVG Trash Cleaner

" mapfile -t LOCS < <(list_locations) # ---------------------------------------------------------------- clean if [ "$ACTION" = "clean" ]; then SAVED="$(cat "$TOKEN_FILE" 2>/dev/null)" rm -f "$TOKEN_FILE" if [ -z "$TOKEN" ] || [ "$TOKEN" != "$SAVED" ]; then echo '

This clean-up link has expired. Please review the trash again before cleaning.

' echo '

Back

' exit 0 fi TOTAL_B=0; TOTAL_N=0; ROWS="" for line in "${LOCS[@]}"; do IFS='|' read -r type label path <<<"$line" read -r before_b before_n <<<"$(loc_stats "$type" "$path")" [ "$before_n" -gt 0 ] || continue empty_location "$type" "$path" read -r after_b after_n <<<"$(loc_stats "$type" "$path")" freed_b=$((before_b - after_b)); freed_n=$((before_n - after_n)) TOTAL_B=$((TOTAL_B + freed_b)); TOTAL_N=$((TOTAL_N + freed_n)) unit="files"; [ "$type" = "mail" ] && unit="messages" ROWS+="$(html_escape "$label")$freed_n $unit$(human_size "$freed_b")" done echo "
Done. Your trash has been emptied.
" echo '
' echo "
$(human_size "$TOTAL_B")
Space cleared
" echo "
$TOTAL_N
Items removed
" echo '
' if [ -n "$ROWS" ]; then echo '' echo "$ROWS" echo '
LocationRemovedCleared
' fi echo '

Mailbox usage figures in DirectAdmin can take a few minutes to update.

' echo '

Scan again

' exit 0 fi # ---------------------------------------------------------------- preview echo '

Empties the File Manager trash and the Trash / Deleted Items folder of every mailbox on your account.

' TOTAL_B=0; TOTAL_N=0; ROWS="" for line in "${LOCS[@]}"; do IFS='|' read -r type label path <<<"$line" read -r b n <<<"$(loc_stats "$type" "$path")" [ "$n" -gt 0 ] || continue TOTAL_B=$((TOTAL_B + b)); TOTAL_N=$((TOTAL_N + n)) unit="files"; [ "$type" = "mail" ] && unit="messages" ROWS+="$(html_escape "$label")$n $unit$(human_size "$b")" done if [ "$TOTAL_N" -eq 0 ]; then echo '
Nothing to clean - all your trash folders are already empty.
' exit 0 fi NEW_TOKEN="$(head -c 24 /dev/urandom | od -An -tx1 | tr -d ' \n')" ( umask 077; printf '%s' "$NEW_TOKEN" > "$TOKEN_FILE" ) echo '
' echo "
$(human_size "$TOTAL_B")
Can be cleared
" echo "
$TOTAL_N
Items in trash
" echo '
' echo '' echo "$ROWS" echo '
LocationItemsSize
' echo "

$BROOM_SVG Clean now

" echo '

Items are deleted permanently and cannot be restored.

' echo ''
